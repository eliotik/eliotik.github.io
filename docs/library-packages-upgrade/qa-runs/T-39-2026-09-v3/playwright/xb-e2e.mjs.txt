// usage: node xb-e2e.mjs <webkit|firefox|chromium> <baseURL> <out.json>
// Analytics safety: the browser uses a local CONNECT proxy (localhost bypassed). The proxy REFUSES (403) the GA
// collect hosts (google-analytics.com, analytics.google.com, doubleclick.net, www.google.com) and tunnels every other
// host to the real internet, so the real gtag.js (googletagmanager.com) and Google Fonts load but no hit reaches GA.
// No request interception (context.route) is used.

const BLOCK_HOST_RE = /(^|\.)(google-analytics\.com|analytics\.google\.com|doubleclick\.net|www\.google\.com)$/i;
const BLOCK_URL_RE = /https?:\/\/([a-z0-9-]+\.)*(google-analytics\.com|analytics\.google\.com|doubleclick\.net|google\.com)\//i;
const SANDBOX_RE = /\/~partytown\/partytown-sandbox-sw\.html/;
const GTAG_RE = /^https:\/\/www\.googletagmanager\.com\/gtag\/js/;
const CAR = '[aria-roledescription="carousel"]';

const check = (id, name, pass, evidence) => {
    R.checks.push({ id: `P9.${engine}.${id}`, name, pass: !!pass, evidence });

// ---- egress proxy ----
const proxy = http.createServer((req, res) => {
    R.proxy.httpRefused.push(`${req.method} ${req.url.slice(0, 120)}`);
    res.writeHead(403);
    res.end();
});
proxy.on('connect', (req, sock, head) => {
    const [host, port] = req.url.split(':');
    if (BLOCK_HOST_RE.test(host)) {
        R.proxy.blocked.push(host);
        sock.end('HTTP/1.1 403 Forbidden\r\n\r\n');
        return;
    }
    R.proxy.tunnels[host] = (R.proxy.tunnels[host] || 0) + 1;
    const up = net.connect(Number(port || 443), host, () => {
        sock.write('HTTP/1.1 200 Connection Established\r\n\r\n');
        if (head && head.length) up.write(head);
        up.pipe(sock);
        sock.pipe(up);
    });
    up.on('error', e => {
        R.proxy.tunnelErrors.push(`${host}: ${e.code || e.message}`);
        sock.destroy();
    });
    sock.on('error', () => up.destroy());
});
await new Promise(r => proxy.listen(0, '127.0.0.1', r));

function instrument(ctx, page, cname) {
    page.on('console', m => {
        const t = m.type();
        if (t === 'error' || t === 'warning') R.console.push({ ctx: cname, src: 'page', label: LABEL, type: t, text: m.text().slice(0, 400), loc: (m.location() && m.location().url) || '' });
    });
    page.on('worker', w => {
        R.workerConsoleHooked = (R.workerConsoleHooked || 0) + 1;
        try {
            w.on('console', m => {
                const t = m.type();
                if (t === 'error' || t === 'warning') R.console.push({ ctx: cname, src: 'worker', label: LABEL, type: t, text: m.text().slice(0, 400), loc: (m.location() && m.location().url) || '', worker: w.url() });
            });
        } catch (e) {
            R.workerConsoleUnsupported = String(e).slice(0, 200);
        }
    });
    page.on('pageerror', e => R.pageErrors.push({ ctx: cname, label: LABEL, text: String(e).slice(0, 400) }));
    ctx.on('requestfailed', r => R.requestFailed.push({ ctx: cname, label: LABEL, url: r.url().slice(0, 160), err: r.failure() && r.failure().errorText }));
}

function ptInitScript() {
    if (window.top !== window) return;
    const S = (window.__pt = { token: Math.random().toString(36).slice(2), iframeLoads: 0, added: 0, removed: 0, pageLoads: 0 });
    const isSb = n => n && n.nodeType === 1 && n.tagName === 'IFRAME' && /\/~partytown\//.test(n.getAttribute('src') || '');
    new MutationObserver(ms => {
        for (const m of ms) {
            for (const n of m.addedNodes) {
                const f = isSb(n) ? n : n.querySelector && n.querySelector('iframe[src*="/~partytown/"]');
                if (f) {
                    S.added++;
                    if (!f.__ptl) {
                        f.__ptl = 1;
                        f.addEventListener('load', () => S.iframeLoads++);
                    }
                }
            }
            for (const n of m.removedNodes) if (isSb(n) || (n.querySelector && n.querySelector('iframe[src*="/~partytown/"]'))) S.removed++;
        }
    }).observe(document, { childList: true, subtree: true });
    document.addEventListener('astro:page-load', () => S.pageLoads++);
}
const snap = page =>
    page.evaluate(() => {
        const f = [...document.querySelectorAll('iframe')].filter(x => /partytown-sandbox-sw\.html/.test(x.getAttribute('src') || ''));
        const ids = window.__ptIds;
        const s1 = document.querySelector('script[data-astro-transition-persist="gtag-src"]');
        return {
            path: location.pathname,
            token: window.__pt.token,
            pageLoads: window.__pt.pageLoads,
            sandboxCount: f.length,
            sandboxParent: f[0] && f[0].parentNode ? f[0].parentNode.nodeName : null,
            iframeSame: ids ? !!f[0] && ids.get(f[0]) === 'iframe' : null,
            gtagSrcSame: ids ? !!s1 && ids.get(s1) === 'gtag-src' : null,
            ptPending: document.querySelectorAll('script[type="text/partytown"]').length,
            iframeLoads: window.__pt.iframeLoads,
            sbAdded: window.__pt.added,
            sbRemoved: window.__pt.removed,
            ssrIslands: document.querySelectorAll('astro-island[ssr]').length,
            islands: document.querySelectorAll('astro-island').length,
        };
    });

// ---- carousel helpers ----
const carState = (page, i) =>
    page.evaluate(
        ({ i, CAR }) => {
            const sc = document.querySelectorAll(CAR)[i];
            const root = sc.parentElement;
            const prev = root.querySelector('button[aria-label="Previous slide"]');
            const next = root.querySelector('button[aria-label="Next slide"]');
            const slides = [...sc.querySelectorAll('[aria-roledescription="slide"]')];
            return {
                scrollLeft: sc.scrollLeft,
                clientWidth: sc.clientWidth,
                maxScroll: sc.scrollWidth - sc.clientWidth,
                prevDisabled: prev ? prev.disabled : null,
                nextDisabled: next ? next.disabled : null,
                N: slides.length,
                labels: slides.map(s => s.getAttribute('aria-label')),
                imgs: slides.map(s => {
                    const im = s.querySelector('img');
                    return im ? { complete: im.complete, nw: im.naturalWidth } : null;
                }),
            };
        },
        { i, CAR }
    );
const settle = (page, i, from) =>
    page.evaluate(
        async ({ i, CAR, from }) => {
            const sc = document.querySelectorAll(CAR)[i];
            const t0 = performance.now();
            while (performance.now() - t0 < 2500 && Math.abs(sc.scrollLeft - from) < 0.5) await new Promise(r => setTimeout(r, 20));
            let last = sc.scrollLeft;
            let stable = 0;
            while (performance.now() - t0 < 8000) {
                await new Promise(r => setTimeout(r, 50));
                if (Math.abs(sc.scrollLeft - last) < 0.01) {
                    if (++stable >= 8) break;
                } else {
                    stable = 0;
                    last = sc.scrollLeft;
                }
            }
            return sc.scrollLeft;
        },
        { i, CAR, from }
    );
async function prepCarousel(page, i) {
    const sc = page.locator(CAR).nth(i);
    await sc.scrollIntoViewIfNeeded();
    await page.waitForFunction(({ i, CAR }) => { const s = document.querySelectorAll(CAR)[i]; return s.querySelectorAll('[aria-roledescription="slide"]').length <= 1 || !s.parentElement.querySelector('button[aria-label="Next slide"]').disabled; }, { i, CAR }, { timeout: 8000 }).catch(() => {});
    return sc.locator('xpath=..');
}
