import { launch, newPage, settle, UP, installVR } from './common.mjs';
const b = await launch();
const p = await newPage(b, 'light', 1280);
const log = [];
p.on('console', (m) => { if (m.type() === 'error') log.push({ phase: 'x', text: m.text(), url: m.location().url }); });
await settle(p, UP, '/this-route-doesnt-exist');
log.forEach((l) => (l.phase = l.phase === 'x' ? 'load' : l.phase));
await p.addStyleTag({ content: '*{transition:none!important}' });
await p.evaluate(installVR, []);
const cdp = await p.context().newCDPSession(p);
await cdp.send('DOM.enable'); await cdp.send('CSS.enable'); await cdp.send('DOM.getDocument', { depth: 0 });
await p.waitForTimeout(500);
log.forEach((l) => (l.phase = l.phase === 'x' ? 'after-addStyleTag/CDP' : l.phase));
console.log(JSON.stringify(log, null, 1));
await b.close();
