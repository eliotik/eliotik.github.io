import fs from 'node:fs';
import path from 'node:path';
const root = process.argv[2];
const files = [];
(function walk(d){ for (const e of fs.readdirSync(d,{withFileTypes:true})) { const p=path.join(d,e.name); if (e.isDirectory()) walk(p); else if (/\.(astro|tsx?|jsx?|mdx?|css)$/.test(e.name)) files.push(p);} })(root);
const counts = {};
const re = /(?<![\w-])((?:[a-z0-9-]+|\[[^\]\s]+\]|group-[a-z-]+|peer-[a-z-]+):)+(?=[!\-\[a-z0-9])/g;
for (const f of files) {
  const t = fs.readFileSync(f,'utf8');
  for (const m of t.matchAll(/class(?:Name|:list)?=\{?["'`]([^"'`]*)["'`]/g)) {
    for (const tok of m[1].split(/\s+/)) {
      const parts = tok.split(':'); if (parts.length<2) continue;
      for (const v of parts.slice(0,-1)) counts[v]=(counts[v]||0)+1;
    }
  }
}
console.log(Object.entries(counts).sort((a,b)=>b[1]-a[1]).map(([k,v])=>`${k}:${v}`).join('  '));
