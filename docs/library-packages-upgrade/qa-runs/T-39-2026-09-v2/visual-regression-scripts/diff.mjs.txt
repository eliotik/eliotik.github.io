const read = (p) => PNG.sync.read(fs.readFileSync(p));

function crop(img, w, h) {
    if (img.width === w && img.height === h) return img;
    const out = new PNG({ width: w, height: h });
    for (let y = 0; y < h; y++)
        img.data.copy(out.data, y * w * 4, y * img.width * 4, y * img.width * 4 + w * 4);
    return out;
}

function boxes(mask, w, h) {
    // mask: Uint8Array w*h of 1 = mismatched; cluster on 16px cells
    const C = 16;
    const cw = Math.ceil(w / C),
        ch = Math.ceil(h / C);
    const cell = new Int32Array(cw * ch).fill(-1);
    const bb = new Map();
    for (let y = 0; y < h; y++)
        for (let x = 0; x < w; x++)
            if (mask[y * w + x]) cell[((y / C) | 0) * cw + ((x / C) | 0)] = 0;
    let id = 0;
    const comps = [];
    for (let i = 0; i < cell.length; i++) {
        if (cell[i] !== 0) continue;
        id++;
        const stack = [i];
        cell[i] = id;
        const cells = [];
        while (stack.length) {
            const c = stack.pop();
            cells.push(c);
            const cx = c % cw,
                cy = (c / cw) | 0;
            for (let dy = -1; dy <= 1; dy++)
                for (let dx = -1; dx <= 1; dx++) {
                    const nx = cx + dx,
                        ny = cy + dy;
                    if (nx < 0 || ny < 0 || nx >= cw || ny >= ch) continue;
                    const n = ny * cw + nx;
                    if (cell[n] === 0) {
                        cell[n] = id;
                        stack.push(n);
                    }
                }
        }
        comps.push(id);
    }
    for (let y = 0; y < h; y++)
        for (let x = 0; x < w; x++) {
            if (!mask[y * w + x]) continue;
            const c = cell[((y / C) | 0) * cw + ((x / C) | 0)];
            const b = bb.get(c) || { x0: x, y0: y, x1: x, y1: y, px: 0 };
            b.x0 = Math.min(b.x0, x);
            b.y0 = Math.min(b.y0, y);
            b.x1 = Math.max(b.x1, x);
            b.y1 = Math.max(b.y1, y);
            b.px++;
            bb.set(c, b);
        }
    return [...bb.values()]
        .sort((a, b) => b.px - a.px)
        .slice(0, 25)
        .map((b) => ({ x: b.x0, y: b.y0, w: b.x1 - b.x0 + 1, h: b.y1 - b.y0 + 1, px: b.px }));
}

export function diffPair(aPath, bPath, { saveDiffTo, saveAbovePct = 0.1 } = {}) {
    const A = read(aPath),
        B = read(bPath);
    const res = {
        a: aPath,
        b: bPath,
        aSize: [A.width, A.height],
        bSize: [B.width, B.height],
        sameSize: A.width === B.width && A.height === B.height,
    };
    const w = Math.min(A.width, B.width),
        h = Math.min(A.height, B.height);
    const a = crop(A, w, h),
        b = crop(B, w, h);
    res.comparedSize = [w, h];
    res.byteIdentical = res.sameSize && Buffer.compare(A.data, B.data) === 0;
    // exact per-pixel RGBA comparison
    let exact = 0,
        firstRow = -1;
    const exactMask = new Uint8Array(w * h);
    for (let i = 0, p = 0; p < w * h; p++, i += 4) {
        if (
            a.data[i] !== b.data[i] ||
            a.data[i + 1] !== b.data[i + 1] ||
            a.data[i + 2] !== b.data[i + 2] ||
            a.data[i + 3] !== b.data[i + 3]
        ) {
            exact++;
            exactMask[p] = 1;
            if (firstRow < 0) firstRow = (p / w) | 0;
        }
    }
    res.exactDiffPixels = exact;
    res.firstDifferingRow = firstRow;
    // pixelmatch threshold 0.1 (task metric), AA excluded (default)
    const out = new PNG({ width: w, height: h });
    const mismatched = pixelmatch(a.data, b.data, out.data, w, h, {
        threshold: 0.1,
    });
    const maskOut = new PNG({ width: w, height: h });
    pixelmatch(a.data, b.data, maskOut.data, w, h, {
        threshold: 0.1,
        diffMask: true,
        diffColor: [255, 0, 0],
        aaColor: [0, 0, 255],
    });
    const pmMask = new Uint8Array(w * h);
    for (let p = 0, i = 0; p < w * h; p++, i += 4)
        if (
            maskOut.data[i] === 255 &&
            maskOut.data[i + 1] === 0 &&
            maskOut.data[i + 2] === 0 &&
            maskOut.data[i + 3] > 0
        )
            pmMask[p] = 1;
    res.pm01Pixels = mismatched;
    res.pm01Pct = +((mismatched / (w * h)) * 100).toFixed(4);
    res.pm0Pixels = pixelmatch(a.data, b.data, null, w, h, {
        threshold: 0,
        includeAA: true,
    });
    res.pm01Boxes = mismatched ? boxes(pmMask, w, h) : [];
    res.exactBoxes = exact ? boxes(exactMask, w, h) : [];
    if (saveDiffTo && res.pm01Pct > saveAbovePct) {
        fs.writeFileSync(saveDiffTo, PNG.sync.write(out));
        res.diffPng = saveDiffTo;
    }
    return res;
}

    // metadata comparison

    const b = capUpB.shots.find((x) => x.file === f);

    const cmp = {};
    for (const k of keys) {
        const uv = JSON.stringify(s[k]).replace(/localhost:4321/g, 'HOST');
        const mv = JSON.stringify(m[k]).replace(/localhost:4331/g, 'HOST');
        if (uv !== mv) cmp[k] = { up: s[k], main: m[k] };
    }
    report.metadata.push({
        file: f,
        differingKeys: Object.keys(cmp),
        diffs: cmp,
        upIframes: s.iframes.map((i) => ({ ...i, src: i.src.replace(/\?\d+$/, '') })),
        mainIframes: m.iframes.map((i) => ({ ...i, src: i.src.replace(/\?\d+$/, '') })),

    });
