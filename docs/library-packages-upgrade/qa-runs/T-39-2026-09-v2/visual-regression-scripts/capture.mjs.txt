// Visual-regression capture: node capture.mjs <baseURL> <outDir> <label> <jsonOut>
import { chromium } from 'playwright';
import fs from 'node:fs';
import path from 'node:path';

const [base, outDir, label, jsonOut] = process.argv.slice(2);
if (!base || !outDir || !label || !jsonOut) {
    console.error('usage: capture.mjs <base> <outDir> <label> <jsonOut>');
    process.exit(2);
}
fs.mkdirSync(outDir, { recursive: true });

export const ROUTES = [
    ['01-home', '/'],
    ['02-posts-index', '/posts'],
    ['03-post-with-images', '/posts/ems-connecting-the-systems'],
    ['04-post-with-carousel', '/posts/flutter-google-maps-embedded-map'],
    ['05-tips-index', '/tips'],
    ['06-tags-index', '/tags'],
    ['07-tag-detail', '/tags/engineering-management/'],
    ['08-threads-index', '/threads'],
    ['09-about', '/about'],
    ['10-404', '/this-route-doesnt-exist'],
    ['11-octoprint', '/posts/octoprint-prusa-core-one-raspberry-pi/'],
    ['12-search', '/search/?q=hiring'],
];
const FULLPAGE = [
    ['fp-home', '/'],
    ['fp-ems-the-delivery-system', '/posts/ems-the-delivery-system/'],
];
const THEMES = ['light', 'dark'];

let browser;
try {
    browser = await chromium.launch({ channel: 'chrome', headless: true });
} catch (e) {
    console.error('channel chrome failed, falling back:', e.message);
    browser = await chromium.launch({
        executablePath:
            '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome',
        headless: true,
    });
}

const result = {
    label,
    base,
    browserVersion: browser.version(),
    startedAt: new Date().toISOString(),
    shots: [],
};

async function capture(name, route, theme, fullPage) {
    const context = await browser.newContext({
        viewport: { width: 1280, height: 1800 },
        deviceScaleFactor: 1,
        reducedMotion: 'reduce',
        colorScheme: theme,
    });
    await context.addInitScript((t) => {
        try {
            localStorage.setItem('theme', t);
        } catch {}
    }, theme);
    const page = await context.newPage();
    const consoleMsgs = [];
    const pageErrors = [];
    const failedRequests = [];
    const badResponses = [];
    page.on('console', (m) => {
        if (['error', 'warning'].includes(m.type()))
            consoleMsgs.push(`${m.type()}: ${m.text()}`);
    });
    page.on('pageerror', (e) => pageErrors.push(String(e)));
    page.on('requestfailed', (r) =>
        failedRequests.push(`${r.url()} :: ${r.failure()?.errorText}`)
    );
    page.on('response', (r) => {
        if (r.status() >= 400 && r.url() !== base + route)
            badResponses.push(`${r.status()} ${r.url()}`);
    });

    const t0 = Date.now();
    const resp = await page.goto(base + route, { waitUntil: 'load' });
    let networkIdle = true;
    try {
        await page.waitForLoadState('networkidle', { timeout: 20000 });
    } catch {
        networkIdle = false;
    }
    await page.evaluate(() => document.fonts.ready);

    let searchReady = null;
    let fpImgsComplete = null;
    if (name === '12-search') {
        try {
            await page.waitForFunction(
                () => /Found \d+ results? for/.test(document.body.innerText),
                null,
                { timeout: 10000 }
            );
            searchReady = true;
        } catch {
            searchReady = false;
        }
    }

    if (fullPage) {
        // trigger lazy images below the fold, then return to top
        const h = await page.evaluate(
            () => document.documentElement.scrollHeight
        );
        for (let y = 0; y < h; y += 900) {
            await page.evaluate((yy) => window.scrollTo(0, yy), y);
            await page.waitForTimeout(120);
        }
        await page.evaluate(() => window.scrollTo(0, 0));
        try {
            await page.waitForLoadState('networkidle', { timeout: 15000 });
        } catch {}
        fpImgsComplete = await page
            .waitForFunction(
                () => [...document.images].every((i) => i.complete),
                null,
                { timeout: 20000 }
            )
            .then(() => true)
            .catch(() => false);
    } else {
        await page
            .waitForFunction(
                () =>
                    [...document.images]
                        .filter((i) => {
                            const r = i.getBoundingClientRect();
                            return (
                                r.top < innerHeight &&
                                r.bottom > 0 &&
                                r.left < innerWidth &&
                                r.right > 0
                            );
                        })
                        .every((i) => i.complete),
                null,
                { timeout: 20000 }
            )
            .catch(() => {});
    }
    await page.evaluate(() => document.fonts.ready);
    await page.waitForTimeout(500);

    const meta = await page.evaluate(() => {
        const de = document.documentElement;
        const cs = getComputedStyle(document.body);
        const imgs = [...document.images];
        const inVp = imgs.filter((i) => {
            const r = i.getBoundingClientRect();
            return (
                r.top < innerHeight &&
                r.bottom > 0 &&
                r.left < innerWidth &&
                r.right > 0
            );
        });
        const iframes = [...document.querySelectorAll('iframe')].map((f) => {
            const r = f.getBoundingClientRect();
            const s = getComputedStyle(f);
            return {
                parent: f.parentElement?.tagName,
                id: f.id,
                src: (f.src || '').replace(location.origin, ''),
                display: s.display,
                visibility: s.visibility,
                rect: [r.x, r.y, r.width, r.height],
            };
        });
        const text = document.body.innerText;
        let h = 0;
        for (let i = 0; i < text.length; i++)
            h = (Math.imul(31, h) + text.charCodeAt(i)) | 0;
        return {
            finalURL: location.href,
            title: document.title,
            dataTheme: de.getAttribute('data-theme'),
            htmlClass: de.className,
            hasDarkClass: de.classList.contains('dark'),
            bodyBg: cs.backgroundColor,
            bodyColor: cs.color,
            bodyFontFamily: cs.fontFamily,
            fontPlexMonoLoaded: document.fonts.check('16px "IBM Plex Mono"'),
            fontsLoaded: [...document.fonts]
                .filter((f) => f.status === 'loaded')
                .map((f) => `${f.family} ${f.weight} ${f.style}`)
                .sort(),
            scrollHeight: de.scrollHeight,
            scrollWidth: de.scrollWidth,
            elementCount: document.getElementsByTagName('*').length,
            imgTotal: imgs.length,
            imgInViewport: inVp.length,
            imgBroken: imgs
                .filter((i) => i.complete && i.naturalWidth === 0)
                .map((i) => i.currentSrc || i.src),
            imgIncompleteAll: imgs
                .filter((i) => !i.complete)
                .map((i) => i.currentSrc || i.src),
            imgInViewportIncomplete: inVp
                .filter((i) => !i.complete)
                .map((i) => i.currentSrc || i.src),
            carouselSlides: document.querySelectorAll(
                '[aria-roledescription="slide"]'
            ).length,
            iframes,
            htmlChildren: [...de.children].map((c) => c.tagName),
            innerTextLen: text.length,
            innerTextHash: h,
            scrollY: window.scrollY,
        };
    });

    const file = `${name}${theme === 'dark' ? '-dark' : ''}.png`;
    await page.screenshot({
        path: path.join(outDir, file),
        fullPage,
        animations: 'disabled',
        caret: 'hide',
    });
    result.shots.push({
        name,
        route,
        theme,
        fullPage,
        file,
        status: resp?.status(),
        networkIdle,
        searchReady,
        fpImgsComplete,
        settleMs: Date.now() - t0,
        ...meta,
        themeOk:
            meta.dataTheme === theme && meta.hasDarkClass === (theme === 'dark'),
        console: consoleMsgs,
        pageErrors,
        failedRequests,
        badResponses,
    });
    console.log(
        `${label} ${file} status=${resp?.status()} theme=${meta.dataTheme}/${meta.hasDarkClass} idle=${networkIdle} imgs=${meta.imgTotal} h=${meta.scrollHeight}`
    );
    await context.close();
}

for (const [name, route] of ROUTES)
    for (const theme of THEMES) await capture(name, route, theme, false);
for (const [name, route] of FULLPAGE)
    for (const theme of THEMES) await capture(name, route, theme, true);

result.finishedAt = new Date().toISOString();
const bad = result.shots.filter(
    (s) =>
        !s.themeOk ||
        (s.name === '12-search' && !s.searchReady) ||
        s.fpImgsComplete === false ||
        s.imgBroken.length ||
        s.imgInViewportIncomplete.length ||
        s.fontPlexMonoLoaded !== true
);
result.problems = bad.map((s) => ({
    file: s.file,
    themeOk: s.themeOk,
    searchReady: s.searchReady,
    fpImgsComplete: s.fpImgsComplete,
    imgBroken: s.imgBroken,
    imgInViewportIncomplete: s.imgInViewportIncomplete,
    fontPlexMonoLoaded: s.fontPlexMonoLoaded,
}));
fs.writeFileSync(jsonOut, JSON.stringify(result, null, 2));
await browser.close();
if (bad.length) {
    console.error('PROBLEMS:', JSON.stringify(result.problems, null, 1));
    process.exit(1);
}
