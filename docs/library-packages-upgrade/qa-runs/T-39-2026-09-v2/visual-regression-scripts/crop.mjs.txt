// node crop.mjs out.png scale x y w h img1 img2 [img3...] -> stacked vertically with 4px magenta separators
import fs from 'node:fs';
import { PNG } from 'pngjs';
const [out, scale, x, y, w, h, ...imgs] = process.argv.slice(2);
const S = +scale, X = +x, Y = +y, W = +w, H = +h;
const res = new PNG({ width: W * S, height: (H * S + 4) * imgs.length });
res.data.fill(0);
imgs.forEach((f, k) => {
  const p = PNG.sync.read(fs.readFileSync(f));
  for (let yy = 0; yy < H * S; yy++) for (let xx = 0; xx < W * S; xx++) {
    const sx = X + Math.floor(xx / S), sy = Y + Math.floor(yy / S);
    const o = ((k * (H * S + 4) + yy) * W * S + xx) * 4;
    if (sx >= p.width || sy >= p.height) { res.data[o] = 128; res.data[o+3] = 255; continue; }
    const i = (sy * p.width + sx) * 4;
    res.data[o] = p.data[i]; res.data[o+1] = p.data[i+1]; res.data[o+2] = p.data[i+2]; res.data[o+3] = 255;
  }
  for (let yy = 0; yy < 4; yy++) for (let xx = 0; xx < W * S; xx++) { const o = ((k * (H * S + 4) + H * S + yy) * W * S + xx) * 4; res.data[o] = 255; res.data[o+2] = 255; res.data[o+3] = 255; }
});
fs.writeFileSync(out, PNG.sync.write(res));
