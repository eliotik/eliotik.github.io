// Characterise the Search.tsx caret race: the component sets selectionStart/End = 0 in a setTimeout(50 ms) after mount.
// Typing that starts inside that window has its first character(s) pushed to the end. Engine-independent (Chrome here).
import { chromium } from 'playwright';
const GA_BLOCK = '--host-resolver-rules=MAP www.google-analytics.com ~NOTFOUND, MAP *.google-analytics.com ~NOTFOUND, MAP analytics.google.com ~NOTFOUND, MAP *.doubleclick.net ~NOTFOUND, MAP www.google.com ~NOTFOUND';
const b = await chromium.launch({ channel: 'chrome', headless: true, args: [GA_BLOCK] });
const out = { browser: b.version(), immediate: [], after300ms: [] };
for (const mode of ['immediate', 'after300ms']) {
    for (let k = 0; k < 5; k++) {
        const ctx = await b.newContext({ viewport: { width: 1280, height: 1800 }, colorScheme: 'light' });
        const page = await ctx.newPage();
        await page.goto('http://localhost:4321/search/', { waitUntil: 'load' });
        await page.waitForFunction(() => document.querySelectorAll('astro-island[ssr]').length === 0);
        if (mode === 'after300ms') await page.waitForTimeout(300);
        await page.locator('input[name="search"]').click();
        await page.keyboard.type('hiring', { delay: 60 });
        await page.waitForTimeout(300);
        out[mode].push(await page.locator('input[name="search"]').inputValue());
        await ctx.close();
    }
}
console.log(JSON.stringify(out));
await b.close();
