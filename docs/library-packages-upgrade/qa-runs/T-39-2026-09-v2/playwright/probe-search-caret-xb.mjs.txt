// QA harness (T-39 Task 8 v2, lane playwright-e2e); not site code. Archived with a .txt suffix so the repo gate
// (`eslint .`, `astro check`) never picks it up. To run: copy to a scratch dir as .mjs next to node_modules/playwright.
// Same as probe-search-caret.mjs, for WebKit/Firefox (Playwright 1.44.1). Egress: dead proxy (localhost bypassed).
// env: BASE (default http://localhost:4321), REPS per mode (default 5).
import pw from 'playwright';
const engine = process.argv[2];
const BASE = (process.env.BASE || 'http://localhost:4321').replace(/\/$/, '');
const REPS = Number(process.env.REPS || 5);
const b = await pw[engine].launch({ headless: true, proxy: { server: 'http://127.0.0.1:9', bypass: 'localhost,127.0.0.1,[::1]' } });
const out = { engine, base: BASE, browser: b.version(), immediate: [], after300ms: [] };
for (const mode of ['immediate', 'after300ms']) {
    for (let k = 0; k < REPS; k++) {
        const ctx = await b.newContext({ viewport: { width: 1280, height: 1800 }, colorScheme: 'light' });
        await ctx.addInitScript(() => {
            window.__caret = [];
            const d = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'selectionStart');
            Object.defineProperty(HTMLInputElement.prototype, 'selectionStart', { configurable: true, get: d.get, set(v) { window.__caret.push({ t: Math.round(performance.now()), v, value: this.value }); d.set.call(this, v); } });
        });
        const page = await ctx.newPage();
        await page.goto(BASE + '/search/', { waitUntil: 'load' });
        await page.waitForFunction(() => document.querySelectorAll('astro-island[ssr]').length === 0);
        const tHyd = await page.evaluate(() => Math.round(performance.now()));
        if (mode === 'after300ms') await page.waitForTimeout(300);
        const tType = await page.evaluate(() => Math.round(performance.now()));
        await page.locator('input[name="search"]').click();
        await page.keyboard.type('hiring', { delay: 60 });
        await page.waitForTimeout(300);
        out[mode].push({ value: await page.locator('input[name="search"]').inputValue(), tHydratedSeen: tHyd, tTypingStart: tType, caretSets: await page.evaluate(() => window.__caret) });
        await ctx.close();
    }
}
console.log(JSON.stringify(out));
await b.close();
