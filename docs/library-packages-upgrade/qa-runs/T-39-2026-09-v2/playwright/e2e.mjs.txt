// Usage: node e2e.mjs <baseURL> <results.json> <screenshotDir>      ONLY=P0,P1,... selects groups.
// Analytics safety: the browser is launched with --host-resolver-rules that NXDOMAIN the GA collect hosts

import { chromium } from 'playwright';
import fs from 'node:fs';
import path from 'node:path';

const BASE = (process.argv[2] || 'http://localhost:4321').replace(/\/$/, '');
const OUT = path.resolve(process.argv[3] || 'results.json');
const SHOTS = path.resolve(process.argv[4] || 'shots');
const IS_PROD = !/localhost|127\.0\.0\.1|\[::1\]/.test(BASE);
fs.mkdirSync(SHOTS, { recursive: true });

const GA_BLOCK =
    '--host-resolver-rules=MAP www.google-analytics.com ~NOTFOUND, MAP *.google-analytics.com ~NOTFOUND, MAP analytics.google.com ~NOTFOUND, MAP *.doubleclick.net ~NOTFOUND, MAP www.google.com ~NOTFOUND';

const CAR = '[aria-roledescription="carousel"]';
const ISLAND_SLIDER = 'astro-island[component-url*="ImageSliderClient"]';
const DESKTOP = { width: 1280, height: 1800 };
const MOBILE = { width: 375, height: 812 };

const results = {
    meta: {
        base: BASE,
        isProd: IS_PROD,
        startedAt: new Date().toISOString(),
        node: process.version,
        gaBlockArg: GA_BLOCK,
    },
    checks: [],
    visits: [],
    console: [],
    requestsFailed: [],
    p0: {},
    details: {},
};

let LABEL = 'init';
let CTX = 'none';
const setLabel = l => {
    LABEL = l;
};
const log = (...a) => console.log(...a);
const sleep = ms => new Promise(r => setTimeout(r, ms));
const slugId = s => s.replace(/[^a-zA-Z0-9_.-]+/g, '_').slice(0, 120);

async function check(page, id, name, pass, evidence) {
    const rec = { id, name, pass: !!pass, evidence };
    if (!pass && page) {
        const f = path.join(SHOTS, `fail-${slugId(id)}.png`);
        try {
            await page.screenshot({ path: f, fullPage: false });
            rec.screenshot = f;
        } catch (e) {
            rec.screenshotError = String(e).slice(0, 200);
        }
    }
    results.checks.push(rec);
    log(`${pass ? 'PASS' : 'FAIL'} ${id} — ${name}`);
    return !!pass;
}

function push(entry) {
    results.console.push({ ctx: CTX, label: LABEL, t: Date.now(), ...entry });
}

function instrument(ctx, name) {
    ctx.on('page', p => attachPage(p, name));
    ctx.on('serviceworker', sw => {
        push({ source: 'sw-created', type: 'info', text: sw.url() });
        try {
            sw.on('console', m =>
                push({ source: 'serviceworker', type: m.type(), text: m.text(), sw: sw.url() })
            );
        } catch {
            /* older API */
        }
    });
    ctx.on('requestfailed', r =>
        results.requestsFailed.push({
            ctx: CTX,
            label: LABEL,
            url: r.url().slice(0, 200),
            type: r.resourceType(),
            err: r.failure()?.errorText,
        })
    );
}

function attachPage(p) {
    p.on('console', m =>
        push({
            source: 'page',
            type: m.type(),
            text: m.text(),
            loc: m.location()?.url,
            url: p.url(),
        })
    );
    p.on('pageerror', e =>
        push({ source: 'pageerror', type: 'error', text: String(e?.stack || e).split('\n').slice(0, 3).join(' | '), url: p.url() })
    );
    p.on('crash', () => push({ source: 'crash', type: 'error', text: 'page crashed', url: p.url() }));
    p.on('worker', w => {
        push({ source: 'worker-created', type: 'info', text: w.url() });
        w.on('console', m => push({ source: 'worker', type: m.type(), text: m.text(), worker: w.url() }));
    });
}

async function newCtx(browser, name, opts = {}) {
    CTX = name;
    const ctx = await browser.newContext({ viewport: DESKTOP, colorScheme: 'light', ...opts });
    instrument(ctx, name);
    return ctx;
}

async function islandsState(page) {
    return page.evaluate(() => ({
        total: document.querySelectorAll('astro-island').length,
        ssr: document.querySelectorAll('astro-island[ssr]').length,
        ssrComponents: [...document.querySelectorAll('astro-island[ssr]')].map(i => i.getAttribute('component-url')),
    }));
}

async function waitHydrated(page, timeout = 15000) {
    await page
        .waitForFunction(() => document.querySelectorAll('astro-island[ssr]').length === 0, null, { timeout })
        .catch(() => {});
    return islandsState(page);
}

async function visit(page, id, how) {
    const hyd = await waitHydrated(page);
    const extra = await page.evaluate(() => ({
        visibility: document.visibilityState,
        title: document.title,
        theme: document.documentElement.getAttribute('data-theme'),
    }));
    const v = { id, ctx: CTX, label: LABEL, url: page.url(), how, islands: hyd.total, islandsSsrRemaining: hyd.ssr, ssrComponents: hyd.ssrComponents, ...extra };
    results.visits.push(v);
    return v;
}

async function go(page, p, id) {
    const resp = await page.goto(BASE + p, { waitUntil: 'load', timeout: 60000 });
    await sleep(300);
    const v = await visit(page, id || p, 'goto');
    v.status = resp?.status();
    return { resp, v };
}

// ---------- carousel helpers ----------
async function carState(page, i) {
    return page.evaluate(
        ({ i, CAR }) => {
            const sc = document.querySelectorAll(CAR)[i];
            if (!sc) return null;
            const root = sc.parentElement;
            const prev = root.querySelector('button[aria-label="Previous slide"]');
            const next = root.querySelector('button[aria-label="Next slide"]');
            const slides = [...sc.querySelectorAll('[aria-roledescription="slide"]')];
            const r = sc.getBoundingClientRect();
            const contentLeft = r.left + sc.clientLeft;
            let best = null;
            slides.forEach((s, k) => {
                const d = s.getBoundingClientRect().left - contentLeft;
                if (!best || Math.abs(d) < Math.abs(best.d)) best = { k: k + 1, d };
            });
            let expectedN = null;
            try {
                expectedN = JSON.parse(sc.closest('astro-island').getAttribute('props')).images[1].length;
            } catch {
                /* ignore */
            }
            const vis = el => {
                if (!el) return false;
                const cs = getComputedStyle(el);
                const b = el.getBoundingClientRect();
                return cs.visibility !== 'hidden' && cs.display !== 'none' && b.width > 0 && b.height > 0;
            };
            return {
                scrollLeft: sc.scrollLeft,
                clientWidth: sc.clientWidth,
                scrollWidth: sc.scrollWidth,
                maxScroll: sc.scrollWidth - sc.clientWidth,
                rectWidth: r.width,
                slideRectWidth: slides[0] ? slides[0].getBoundingClientRect().width : null,
                prevDisabled: prev ? prev.disabled : null,
                nextDisabled: next ? next.disabled : null,
                prevVisible: vis(prev),
                nextVisible: vis(next),
                prevOpacity: prev ? getComputedStyle(prev).opacity : null,
                nextOpacity: next ? getComputedStyle(next).opacity : null,
                slideCount: slides.length,
                expectedN,
                labels: slides.map(s => s.getAttribute('aria-label')),
                nearestSlide: best ? best.k : null,
                nearestOffset: best ? Math.round(best.d * 1000) / 1000 : null,
                imgs: slides.map(s => {
                    const im = s.querySelector('img');
                    return im ? { complete: im.complete, nw: im.naturalWidth, src: im.getAttribute('src') } : null;
                }),
                scrollY: window.scrollY,
                activeIsNext: document.activeElement === next,
            };
        },
        { i, CAR }
    );
}

async function armTracker(page, i) {
    await page.evaluate(
        ({ i, CAR }) => {
            const sc = document.querySelectorAll(CAR)[i];
            if (window.__trkEl && window.__trkHandler) window.__trkEl.removeEventListener('scroll', window.__trkHandler);
            const t = { start: sc.scrollLeft, min: sc.scrollLeft, max: sc.scrollLeft, events: 0 };
            window.__trk = t;
            window.__trkEl = sc;
            window.__trkHandler = () => {
                t.events++;
                t.min = Math.min(t.min, sc.scrollLeft);
                t.max = Math.max(t.max, sc.scrollLeft);
            };
            sc.addEventListener('scroll', window.__trkHandler, { passive: true });
        },
        { i, CAR }
    );
}

async function settle(page, i, expectMove = true) {
    return page.evaluate(
        async ({ i, CAR, expectMove }) => {
            const sc = document.querySelectorAll(CAR)[i];
            const t0 = performance.now();
            const start = window.__trk ? window.__trk.start : sc.scrollLeft;
            if (expectMove) {
                while (performance.now() - t0 < 2500 && Math.abs(sc.scrollLeft - start) < 0.5 && (window.__trk ? window.__trk.events : 0) === 0)
                    await new Promise(r => setTimeout(r, 20));
            }
            let last = sc.scrollLeft;
            let stable = 0;
            while (performance.now() - t0 < 8000) {
                await new Promise(r => setTimeout(r, 50));
                if (Math.abs(sc.scrollLeft - last) < 0.01) {
                    if (++stable >= 8) break;
                } else {
                    stable = 0;
                    last = sc.scrollLeft;
                }
            }
            await new Promise(r => requestAnimationFrame(() => requestAnimationFrame(r)));
            return { settledMs: Math.round(performance.now() - t0), trk: window.__trk ? { ...window.__trk } : null };
        },
        { i, CAR, expectMove }
    );
}

const snapInfo = st => {
    const cw = st.clientWidth;
    const mod = ((st.scrollLeft % cw) + cw) % cw;
    const modDist = Math.min(mod, cw - mod);
    return {
        mod: Math.round(mod * 1000) / 1000,
        modDist: Math.round(modDist * 1000) / 1000,
        modOk: modDist <= 2,
        geomOffset: st.nearestOffset,
        geomOk: Math.abs(st.nearestOffset) <= 2,
        nearestSlide: st.nearestSlide,
    };
};

async function press(page, locator, mobile) {
    if (mobile) await locator.tap({ timeout: 5000 });
    else await locator.click({ timeout: 5000 });
}

// swipe via CDP touch gesture (mobile) or horizontal wheel (desktop). dir: +1 = towards next slide.
async function gesture(page, cdp, sc, frac, dir, mobile) {
    const box = await sc.boundingBox();
    const cx = box.x + box.width / 2;
    const cy = box.y + box.height / 2;
    const st = await sc.evaluate(el => el.clientWidth);
    const dist = Math.round(frac * st);
    if (mobile) {
        // synthesizeScrollGesture: positive xDistance scrolls content towards the left edge (scrollLeft decreases).
        await cdp.send('Input.synthesizeScrollGesture', {
            x: Math.round(cx),
            y: Math.round(cy),
            xDistance: -dir * dist,
            yDistance: 0,
            gestureSourceType: 'touch',
            speed: 600,
            preventFling: true,
        });
        return { kind: 'touch-synthesizeScrollGesture', xDistance: -dir * dist, at: [Math.round(cx), Math.round(cy)] };
    }
    await page.mouse.move(cx, cy);
    await page.mouse.wheel(dir * dist, 0);
    return { kind: 'mouse.wheel', deltaX: dir * dist, at: [Math.round(cx), Math.round(cy)] };
}

async function sliderSuite(page, slug, prefix, { mobile = false, cdp = null } = {}) {
    setLabel(`${prefix}:${slug}`);
    const { v } = await go(page, `/posts/${slug}/`, `${prefix}:${slug}`);
    const nIslands = await page.locator(ISLAND_SLIDER).count();
    const n = await page.locator(CAR).count();
    await check(page, `${prefix}.${slug}.carousel-count`, `carousel count on ${slug}`, n > 0 && n === nIslands, { carousels: n, sliderIslands: nIslands, status: v.status });
    const perCar = [];
    for (let i = 0; i < n; i++) {
        const id = `${prefix}.${slug}.c${i + 1}`;
        const rec = { index: i + 1, steps: [] };
        perCar.push(rec);
        try {
            const sc = page.locator(CAR).nth(i);
            const root = sc.locator('xpath=..');
            const next = root.locator('button[aria-label="Next slide"]');
            const prev = root.locator('button[aria-label="Previous slide"]');
            await sc.scrollIntoViewIfNeeded();
            await sleep(150);
            // wait post-mount state (useEffect -> updateScrollState)
            await page
                .waitForFunction(
                    ({ i, CAR }) => {
                        const s = document.querySelectorAll(CAR)[i];
                        const nb = s.parentElement.querySelector('button[aria-label="Next slide"]');
                        return s.querySelectorAll('[aria-roledescription="slide"]').length <= 1 || !nb.disabled;
                    },
                    { i, CAR },
                    { timeout: 5000 }
                )
                .catch(() => {});
            const st0 = await carState(page, i);
            rec.start = st0;
            const N = st0.slideCount;
            const cw = st0.clientWidth;
            const expLabels = Array.from({ length: N }, (_, k) => `Slide ${k + 1} of ${N}`);
            await check(page, `${id}.slide-count`, `slide count matches island props (N)`, N > 0 && N === st0.expectedN, { N, expectedNFromProps: st0.expectedN });
            await check(page, `${id}.labels`, `labels 'Slide k of N'`, JSON.stringify(st0.labels) === JSON.stringify(expLabels), { labels: st0.labels });
            rec.imgsLoadedAtStart = st0.imgs.filter(x => x && x.complete && x.nw > 0).length;
            if (N < 2) {
                await check(page, `${id}.single`, 'single-slide carousel: both buttons disabled', st0.prevDisabled && st0.nextDisabled, st0);
                continue;
            }
            await check(page, `${id}.start`, 'start: scrollLeft 0, Prev disabled, Next enabled', st0.scrollLeft <= 1 && st0.prevDisabled === true && st0.nextDisabled === false && st0.nextVisible, {
                scrollLeft: st0.scrollLeft,
                prevDisabled: st0.prevDisabled,
                nextDisabled: st0.nextDisabled,
                prevOpacity: st0.prevOpacity,
                nextVisible: st0.nextVisible,
                clientWidth: cw,
                scrollWidth: st0.scrollWidth,
                slideRectWidth: st0.slideRectWidth,
            });
            // --- Next click (real mouse click / tap) ---
            const stepDeltas = [];
            let cur = st0;
            await armTracker(page, i);
            await press(page, next, mobile);
            let s = await settle(page, i);
            let st = await carState(page, i);
            const d1 = st.scrollLeft - cur.scrollLeft;
            stepDeltas.push(d1);
            rec.steps.push({ action: mobile ? 'tap Next' : 'click Next', from: cur.scrollLeft, to: st.scrollLeft, delta: d1, settledMs: s.settledMs, snap: snapInfo(st) });
            await check(page, `${id}.next-click`, `${mobile ? 'tap' : 'mouse click'} Next advances scrollLeft by clientWidth ±2`, Math.abs(d1 - cw) <= 2 && st.nearestSlide === 2, {
                from: cur.scrollLeft,
                to: st.scrollLeft,
                delta: d1,
                clientWidth: cw,
                nearestSlide: st.nearestSlide,
                prevDisabledAfter: st.prevDisabled,
            });
            cur = st;
            // --- walk to end ---
            let guard = 0;
            while (!cur.nextDisabled && guard++ < N + 2) {
                await armTracker(page, i);
                await press(page, next, mobile);
                s = await settle(page, i);
                st = await carState(page, i);
                stepDeltas.push(st.scrollLeft - cur.scrollLeft);
                rec.steps.push({ action: mobile ? 'tap Next' : 'click Next', from: cur.scrollLeft, to: st.scrollLeft, delta: st.scrollLeft - cur.scrollLeft, settledMs: s.settledMs, snap: snapInfo(st) });
                cur = st;
            }
            await check(page, `${id}.walk-end`, `walk to end: ${N - 1} Next presses, every step = clientWidth ±2, Next disabled at end`, stepDeltas.length === N - 1 && stepDeltas.every(d => Math.abs(d - cw) <= 2) && cur.nextDisabled === true && cur.prevDisabled === false && Math.abs(cur.scrollLeft - cur.maxScroll) <= 2 && cur.nearestSlide === N, {
                presses: stepDeltas.length,
                stepDeltas,
                endScrollLeft: cur.scrollLeft,
                maxScroll: cur.maxScroll,
                nextDisabled: cur.nextDisabled,
                prevDisabled: cur.prevDisabled,
                nearestSlide: cur.nearestSlide,
            });
            // --- Prev once ---
            await armTracker(page, i);
            await press(page, prev, mobile);
            s = await settle(page, i);
            st = await carState(page, i);
            const dp = st.scrollLeft - cur.scrollLeft;
            rec.steps.push({ action: mobile ? 'tap Prev' : 'click Prev', from: cur.scrollLeft, to: st.scrollLeft, delta: dp, snap: snapInfo(st) });
            await check(page, `${id}.prev-click`, 'Prev goes back exactly one slide (−clientWidth ±2), Next re-enabled', Math.abs(dp + cw) <= 2 && st.nearestSlide === N - 1 && st.nextDisabled === false, {
                from: cur.scrollLeft,
                to: st.scrollLeft,
                delta: dp,
                nearestSlide: st.nearestSlide,
                nextDisabled: st.nextDisabled,
            });
            cur = st;
            // --- keyboard: Enter on focused Next ---
            await next.focus();
            const fEnter = await carState(page, i);
            await armTracker(page, i);
            await page.keyboard.press('Enter');
            s = await settle(page, i);
            st = await carState(page, i);
            const dk = st.scrollLeft - cur.scrollLeft;
            rec.steps.push({ action: 'keyboard Enter on focused Next', from: cur.scrollLeft, to: st.scrollLeft, delta: dk, scrollYBefore: fEnter.scrollY, scrollYAfter: st.scrollY });
            await check(page, `${id}.kbd-enter`, 'focus Next + Enter advances one slide', fEnter.activeIsNext && Math.abs(dk - cw) <= 2 && st.nearestSlide === N, {
                focusedNextBeforePress: fEnter.activeIsNext,
                delta: dk,
                nearestSlide: st.nearestSlide,
                scrollYBefore: fEnter.scrollY,
                scrollYAfter: st.scrollY,
            });
            cur = st;
            // back one with Prev, then Space
            await armTracker(page, i);
            await press(page, prev, mobile);
            await settle(page, i);
            cur = await carState(page, i);
            await next.focus();
            const fSpace = await carState(page, i);
            await armTracker(page, i);
            await page.keyboard.press('Space');
            s = await settle(page, i);
            st = await carState(page, i);
            const ds = st.scrollLeft - cur.scrollLeft;
            rec.steps.push({ action: 'keyboard Space on focused Next', from: cur.scrollLeft, to: st.scrollLeft, delta: ds, scrollYBefore: fSpace.scrollY, scrollYAfter: st.scrollY });
            await check(page, `${id}.kbd-space`, 'focus Next + Space advances one slide, page does not scroll', fSpace.activeIsNext && Math.abs(ds - cw) <= 2 && st.nearestSlide === N && Math.abs(st.scrollY - fSpace.scrollY) < 1, {
                focusedNextBeforePress: fSpace.activeIsNext,
                prevBeforeSpace: cur.nearestSlide,
                delta: ds,
                nearestSlide: st.nearestSlide,
                scrollYBefore: fSpace.scrollY,
                scrollYAfter: st.scrollY,
            });
            cur = st;
            // --- images all loaded (every slide visited) ---
            await page
                .waitForFunction(
                    ({ i, CAR }) => [...document.querySelectorAll(CAR)[i].querySelectorAll('img')].every(im => im.complete && im.naturalWidth > 0),
                    { i, CAR },
                    { timeout: 10000 }
                )
                .catch(() => {});
            const stImg = await carState(page, i);
            await check(page, `${id}.images`, 'every slide <img> complete && naturalWidth > 0', stImg.imgs.every(x => x && x.complete && x.nw > 0), {
                loadedAtStart: rec.imgsLoadedAtStart,
                imgs: stImg.imgs.map(x => ({ src: x?.src, nw: x?.nw, complete: x?.complete })),
            });
            // --- gesture (wheel / touch) from the end, towards previous (partial slide) ---
            await armTracker(page, i);
            const g1 = await gesture(page, cdp, sc, 0.4, -1, mobile);
            s = await settle(page, i);
            st = await carState(page, i);
            const sn1 = snapInfo(st);
            const moved1 = s.trk ? Math.max(s.trk.start - s.trk.min, s.trk.max - s.trk.start) : 0;
            rec.steps.push({ action: `${g1.kind} 40% back from end`, gesture: g1, from: cur.scrollLeft, to: st.scrollLeft, maxDisplacement: moved1, snap: sn1, trk: s.trk });
            await check(page, `${id}.${mobile ? 'swipe' : 'wheel'}-back`, `${mobile ? 'touch swipe' : 'horizontal wheel'} of 40% slide (backwards from end): scroller moved and settled snapped (scrollLeft % clientWidth ≈ 0)`, moved1 > 5 && (sn1.modOk || sn1.geomOk) && sn1.geomOk, {
                gesture: g1,
                from: cur.scrollLeft,
                settledAt: st.scrollLeft,
                maxDisplacementDuringGesture: moved1,
                scrollEvents: s.trk?.events,
                modDist: sn1.modDist,
                geomOffset: sn1.geomOffset,
                landedOnSlide: sn1.nearestSlide,
            });
            cur = st;
            // --- walk back to start with Prev ---
            guard = 0;
            const backDeltas = [];
            while (!cur.prevDisabled && guard++ < N + 2) {
                await armTracker(page, i);
                await press(page, prev, mobile);
                await settle(page, i);
                st = await carState(page, i);
                backDeltas.push(st.scrollLeft - cur.scrollLeft);
                cur = st;
            }
            await check(page, `${id}.walk-start`, 'Prev back to start: each step −clientWidth ±2, Prev disabled at start, Next enabled', backDeltas.every(d => Math.abs(d + cw) <= 2) && cur.scrollLeft <= 1 && cur.prevDisabled === true && cur.nextDisabled === false, {
                backDeltas,
                scrollLeft: cur.scrollLeft,
                prevDisabled: cur.prevDisabled,
                nextDisabled: cur.nextDisabled,
            });
            // --- gesture forward from start (partial slide) ---
            await armTracker(page, i);
            const g2 = await gesture(page, cdp, sc, 0.4, +1, mobile);
            s = await settle(page, i);
            st = await carState(page, i);
            const sn2 = snapInfo(st);
            const moved2 = s.trk ? Math.max(s.trk.start - s.trk.min, s.trk.max - s.trk.start) : 0;
            rec.steps.push({ action: `${g2.kind} 40% forward from start`, gesture: g2, from: cur.scrollLeft, to: st.scrollLeft, maxDisplacement: moved2, snap: sn2, trk: s.trk });
            await check(page, `${id}.${mobile ? 'swipe' : 'wheel'}-fwd`, `${mobile ? 'touch swipe' : 'horizontal wheel'} of 40% slide (forward from start): scroller moved and settled snapped`, moved2 > 5 && sn2.geomOk, {
                gesture: g2,
                from: cur.scrollLeft,
                settledAt: st.scrollLeft,
                maxDisplacementDuringGesture: moved2,
                scrollEvents: s.trk?.events,
                modDist: sn2.modDist,
                geomOffset: sn2.geomOffset,
                landedOnSlide: sn2.nearestSlide,
                buttonsAfter: { prevDisabled: st.prevDisabled, nextDisabled: st.nextDisabled },
            });
            // button state consistent with the position after a gesture
            const consistent = st.prevDisabled === st.scrollLeft <= 0 && st.nextDisabled === !(st.scrollLeft + st.clientWidth < st.scrollWidth - 1);
            await check(page, `${id}.buttons-track-gesture`, 'Prev/Next disabled state tracks the position after a gesture', consistent, { scrollLeft: st.scrollLeft, prevDisabled: st.prevDisabled, nextDisabled: st.nextDisabled });
            cur = st;
            {
                // a longer gesture (70% of a slide) must advance to the next slide and snap
                await armTracker(page, i);
                const g3 = await gesture(page, cdp, sc, 0.7, +1, mobile);
                s = await settle(page, i);
                st = await carState(page, i);
                const sn3 = snapInfo(st);
                rec.steps.push({ action: `${g3.kind} 70% forward`, gesture: g3, from: cur.scrollLeft, to: st.scrollLeft, snap: sn3, trk: s.trk });

                    gesture: g3,
                    maxDisplacementDuringGesture: s.trk ? Math.max(s.trk.start - s.trk.min, s.trk.max - s.trk.start) : null,
                    from: cur.scrollLeft,
                    fromSlide: cur.nearestSlide,
                    settledAt: st.scrollLeft,
                    landedOnSlide: st.nearestSlide,
                    geomOffset: sn3.geomOffset,

                });
            }
        } catch (e) {

        }
    }
    return perCar;
}

// parse any CSS color via canvas -> [r,g,b,a]
async function colorProbe(page, i) {
    return page.evaluate(
        ({ i, CAR }) => {
            const cv = document.createElement('canvas');
            cv.width = cv.height = 1;
            const c2 = cv.getContext('2d', { willReadFrequently: true });
            const toRGBA = col => {
                c2.clearRect(0, 0, 1, 1);
                c2.fillStyle = '#000';
                c2.fillStyle = col;
                c2.fillRect(0, 0, 1, 1);
                const d = c2.getImageData(0, 0, 1, 1).data;
                return [d[0], d[1], d[2], d[3] / 255];
            };
            const lum = ([r, g, b]) => {
                const f = v => {
                    v /= 255;
                    return v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4;
                };
                return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b);
            };
            const over = (fg, bg) => [0, 1, 2].map(k => fg[k] * fg[3] + bg[k] * (1 - fg[3]));
            const sc = document.querySelectorAll(CAR)[i];
            const next = sc.parentElement.querySelector('button[aria-label="Next slide"]');
            const bodyBg = toRGBA(getComputedStyle(document.body).backgroundColor);
            const btnBg = toRGBA(getComputedStyle(next).backgroundColor);
            const btnFg = toRGBA(getComputedStyle(next).color);
            const btnBgComposite = over(btnBg, bodyBg);
            const L1 = lum(btnFg);
            const L2 = lum(btnBgComposite);
            const ratio = (Math.max(L1, L2) + 0.05) / (Math.min(L1, L2) + 0.05);
            return {
                theme: document.documentElement.getAttribute('data-theme'),
                bodyBg: getComputedStyle(document.body).backgroundColor,
                bodyBgRGBA: bodyBg,
                btnBg: getComputedStyle(next).backgroundColor,
                btnFg: getComputedStyle(next).color,
                btnFgRGBA: btnFg,
                btnBgCompositeRGB: btnBgComposite.map(Math.round),
                iconVsButtonContrast: Math.round(ratio * 100) / 100,
                scrollerBorderColor: getComputedStyle(sc).borderTopColor,
                bodyLuminance: Math.round(lum(bodyBg) * 1000) / 1000,
            };
        },
        { i, CAR }
    );
}

function ptInitScript() {
    if (window.top !== window) return; // init scripts also run in the sandbox iframe
    const S = (window.__pt = { token: Math.random().toString(36).slice(2), iframeLoads: [], added: 0, removed: 0, pageLoads: 0, beforeSwaps: 0, afterSwaps: 0 });
    const isSb = n => n && n.nodeType === 1 && n.tagName === 'IFRAME' && /\/~partytown\//.test(n.getAttribute('src') || '');
    new MutationObserver(ms => {
        for (const m of ms) {
            for (const n of m.addedNodes) {
                const f = isSb(n) ? n : n.querySelector && n.querySelector('iframe[src*="/~partytown/"]');
                if (f) {
                    S.added++;
                    if (!f.__ptl) {
                        f.__ptl = 1;
                        f.addEventListener('load', () => S.iframeLoads.push({ t: Math.round(performance.now()), path: location.pathname }));
                    }
                }
            }
            for (const n of m.removedNodes) if (isSb(n) || (n.querySelector && n.querySelector('iframe[src*="/~partytown/"]'))) S.removed++;
        }
    }).observe(document, { childList: true, subtree: true });
    document.addEventListener('astro:page-load', () => S.pageLoads++);
    document.addEventListener('astro:before-swap', () => S.beforeSwaps++);
    document.addEventListener('astro:after-swap', () => S.afterSwaps++);
}

const ptSnap = page =>
    page.evaluate(() => {

        const s1 = document.querySelector('script[data-astro-transition-persist="gtag-src"]');
        const s2 = document.querySelector('script[data-astro-transition-persist="gtag-init"]');
        const ids = window.__ptIds;
        return {
            path: location.pathname,
            token: window.__pt.token,
            pageLoads: window.__pt.pageLoads,
            beforeSwaps: window.__pt.beforeSwaps,
            sandboxCount: f.length,
            sandboxParent: f[0] && f[0].parentNode ? f[0].parentNode.nodeName : null,
            sandboxIsLastChildOfHtml: f[0] ? f[0].parentNode === document.documentElement : null,
            iframeSameElement: ids ? !!f[0] && ids.get(f[0]) === 'iframe' : null,
            gtagSrcSameElement: ids ? !!s1 && ids.get(s1) === 'gtag-src' : null,
            gtagInitSameElement: ids ? !!s2 && ids.get(s2) === 'gtag-init' : null,
            gtagScriptTypes: [s1 && s1.type, s2 && s2.type],
            ptPending: document.querySelectorAll('script[type="text/partytown"]').length,
            iframeLoads: window.__pt.iframeLoads.length,
            sbAdded: window.__pt.added,

    });

function parseCollect(req) {
    const u = new URL(req.url());
    const body = (() => {
        try {
            return req.postData() || '';
        } catch {
            return '';
        }
    })();
    const lines = body.split(/\r?\n/).filter(l => l.includes('='));
    return (lines.length ? lines : ['']).map(l => {
        const e = { ...Object.fromEntries(u.searchParams), ...Object.fromEntries(new URLSearchParams(l)) };
        return { en: e.en, dl: (e.dl || '').replace(/^https?:\/\/[^/]+/, ''), host: u.host };
    });
}

async function tagsState(page) {
    return page.evaluate(() => {
        const items = [...document.querySelectorAll('astro-island ul li a')].map(a => {
            const spans = a.querySelectorAll(':scope > span');
            return { tag: a.getAttribute('href').replace(/^\/tags\//, '').replace(/\/$/, ''), count: Number(spans[2]?.textContent.trim()) };
        });
        const btns = [...document.querySelectorAll('astro-island button[aria-pressed]')].map(b => ({ text: b.textContent.trim(), pressed: b.getAttribute('aria-pressed') }));
        const summary = [...document.querySelectorAll('astro-island div')].map(d => d.textContent.trim()).find(t => /^\d+ of \d+ tags/.test(t)) || null;
        return { items, btns, summary };
    });
}
const isCountSorted = items => items.every((t, k) => k === 0 || items[k - 1].count > t.count || (items[k - 1].count === t.count && items[k - 1].tag.localeCompare(t.tag) <= 0));
const isAlphaSorted = items => items.every((t, k) => k === 0 || items[k - 1].tag.localeCompare(t.tag) <= 0);

// the carousel post is linked from home on the branch/main; the older production deploy may not have it
async function postLink(page) {
    const pref = 'main a[href="/posts/octoprint-prusa-core-one-raspberry-pi/"]';
    const sel = (await page.locator(pref).count()) > 0 ? pref : 'main a[href^="/posts/"]:not([href="/posts/"])';
    const href = await page.locator(sel).first().getAttribute('href');
    return { loc: page.locator(sel).first(), href, fallback: sel !== pref };
}

        const openAfter = await det.first().evaluate(d => d.open);
        const firstLinkVisibleOpen = await det.first().locator('a').first().isVisible();
        const anchors = await page.evaluate(links => links.map(h => ({ href: h, exists: h.startsWith('#') && !!document.getElementById(decodeURIComponent(h.slice(1))) })), s0.links);
        await check(page, `P6.${slug}.collapse`, `TOC <details> closed initially, opens on summary click`, cnt === 1 && s0.open === false && !firstLinkVisibleClosed && openAfter === true && firstLinkVisibleOpen, { detailsCount: cnt, summary: s0.summary, openInitially: s0.open, linkVisibleWhenClosed: firstLinkVisibleClosed, openAfterClick: openAfter, linkVisibleWhenOpen: firstLinkVisibleOpen });
        await check(page, `P6.${slug}.anchors`, 'every TOC link targets an existing in-page id', anchors.length > 0 && anchors.every(a => a.exists), { links: anchors.length, missing: anchors.filter(a => !a.exists) });
        // click two links for real
        const clicked = [];
        for (const idx of [1, anchors.length - 1]) {
            const href = s0.links[idx];
            await det.first().locator('a').nth(idx).click();
            await page.waitForFunction(h => location.hash === h, href, { timeout: 5000 }).catch(() => {});
            // wait for (smooth) window scroll to settle
            await page.evaluate(async () => {
                let last = scrollY;
                let stable = 0;
                const t0 = performance.now();
                while (performance.now() - t0 < 5000) {
                    await new Promise(r => setTimeout(r, 50));
                    if (scrollY === last) {
                        if (++stable >= 8) break;
                    } else {
                        stable = 0;
                        last = scrollY;
                    }
                }
            });
            const r = await page.evaluate(h => {
                const el = document.getElementById(decodeURIComponent(h.slice(1)));
                const top = el.getBoundingClientRect().top;
                return { hash: location.hash, path: location.pathname, targetTop: Math.round(top), scrollY: Math.round(scrollY), atBottom: Math.ceil(scrollY + innerHeight) >= document.documentElement.scrollHeight - 2 };
            }, href);
            clicked.push({ href, ...r });
        }
        await check(page, `P6.${slug}.navigate`, 'clicking TOC links scrolls to the in-page anchor (hash set, target in viewport)', clicked.every(c => c.hash === c.href && c.path === `/posts/${slug}/` && c.targetTop >= -2 && (c.targetTop < 300 || c.atBottom)), { clicked });
        results.details.P6[slug] = { s0, anchors, clicked };
    }
    await ctx.close();
}

// P8 (extra): the v1 proxytown characterisation flow (home -> post -> search) at 0.5/3/10 s dwell, 2 reps each,
// fresh context per run. After the Task 14 fix every run must show 0 console errors and 0 aborted proxytown XHRs.
async function runP8() {
    results.details.P8 = [];
    for (const dwell of [500, 3000, 10000]) {
        for (let rep = 1; rep <= 2; rep++) {
            const name = `p8-dwell${dwell}-r${rep}`;
            const ctx = await newCtx(blockedBrowser, name);
            const page = await ctx.newPage();
            setLabel(`P8:${dwell}:${rep}:home`);
            await page.goto(BASE + '/', { waitUntil: 'load' });
            await sleep(dwell);
            setLabel(`P8:${dwell}:${rep}:post`);
            const pl = await postLink(page);
            await pl.loc.click();
            await page.waitForFunction(h => location.pathname === h, pl.href, { timeout: 15000 }).catch(() => {});
            await sleep(dwell);
            setLabel(`P8:${dwell}:${rep}:search`);
            await page.locator('header a[href="/search/"]').click();
            await page.waitForFunction(() => location.pathname === '/search/', null, { timeout: 15000 }).catch(() => {});
            await sleep(3000);
            await visit(page, `P8:${dwell}:${rep}:/search/`, 'client-nav');
            const errs = results.console.filter(c => c.ctx === name && (c.type === 'error' || c.source === 'pageerror' || c.source === 'crash'));
            const row = {
                dwellMs: dwell,
                rep,
                path: new URL(page.url()).pathname,
                errors: errs.length,
                proxytownErrors: errs.filter(e => /proxytown/.test(e.text)).length,
                errorTexts: errs.map(e => e.text.split('\n')[0]),
                abortedProxytownXHR: results.requestsFailed.filter(r => r.ctx === name && /proxytown/.test(r.url)).length,
            };
            results.details.P8.push(row);
            await check(page, `P8.dwell${dwell}.r${rep}`, `home → post → search at ${dwell} ms dwell: 0 console errors, 0 aborted proxytown XHRs`, row.path === '/search/' && row.errors === 0 && row.abortedProxytownXHR === 0, row);
            await ctx.close();
        }
    }
}

// H: harness positive control. Proves the capture pipeline sees page console.error, a dedicated-worker
// console.error, an uncaught exception (pageerror) and a 404 resource error. Excluded from the zero criterion.
async function runH() {
    const ctx = await newCtx(blockedBrowser, 'h-control');
    const page = await ctx.newPage();
    setLabel('H:control');
    await page.goto(BASE + '/about/', { waitUntil: 'load' });
    await page.evaluate(async () => {
        console.error('HARNESS-CONTROL page console.error');
        const w = new Worker(URL.createObjectURL(new Blob(["console.error('HARNESS-CONTROL worker console.error')"], { type: 'text/javascript' })));
        setTimeout(() => {
            throw new Error('HARNESS-CONTROL uncaught');
        }, 0);
        const img = new Image();
        img.src = '/harness-control-missing-' + Date.now() + '.png';
        document.body.appendChild(img);
        await new Promise(r => setTimeout(r, 1500));
        w.terminate();
    });
    await sleep(500);
    const got = results.console.filter(c => c.ctx === 'h-control');
    const has = (src, re) => got.some(c => c.source === src && re.test(c.text || ''));
    const r = {
        pageConsoleError: has('page', /HARNESS-CONTROL page console\.error/),
        workerConsoleError: has('worker', /HARNESS-CONTROL worker console\.error/) || has('page', /HARNESS-CONTROL worker console\.error/),
        pageerror: has('pageerror', /HARNESS-CONTROL uncaught/),
        resource404: got.some(c => c.type === 'error' && /404/.test(c.text || '')),
    };
    await check(page, 'H.capture-control', 'harness positive control: page console.error, worker console.error, pageerror and a 404 resource error are all captured', Object.values(r).every(Boolean), { ...r, captured: got.map(c => `${c.source}:${c.type}:${(c.text || '').slice(0, 120)}`) });
    await ctx.close();
}
