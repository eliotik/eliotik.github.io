import { PNG } from 'pngjs'; import fs from 'node:fs';
const [,, src, out, y0, y1] = process.argv;
const a = PNG.sync.read(fs.readFileSync(src));
// find diff columns bbox if a diff image companion exists
const Y0 = +y0, Y1 = +y1, H = Y1 - Y0;
const o = new PNG({ width: a.width, height: H });
for (let y = 0; y < H; y++) a.data.copy(o.data, y * a.width * 4, (Y0 + y) * a.width * 4, (Y0 + y + 1) * a.width * 4);
fs.writeFileSync(out, PNG.sync.write(o));
